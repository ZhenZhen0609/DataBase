说明：

2. 请先执行“环境准备”部分中的语句，再按顺序运行各测试用例。
3. 每条用例下方用注释标注预期结果，实际执行时请对照日志输出或查询结果。

================================================================================
一、环境准备
================================================================================

-- 1. 创建数据库
CREATE DATABASE DMLTestDB;

-- 2. 选择数据库（在 GUI 中点击左侧数据库，或在代码中设置当前数据库）
-- 注意：如果当前没有自动切换，请在 UI 或代码中设置当前数据库为 DMLTestDB

-- 3. 创建测试表
CREATE TABLE students (id INT, name TEXT, age INT, score DOUBLE);

-- 4. 批量插入初始数据
INSERT INTO students VALUES 
(1, 'Alice', 20, 88.5),
(2, 'Bob', 22, 76.0),
(3, 'Charlie', 19, 92.0),
(4, 'Diana', 21, 85.5),
(5, 'Alice', 23, 90.0);

-- 预期：插入 5 条记录，日志显示 "Inserted 5 rows"

================================================================================
二、基础 SELECT 查询
================================================================================

-- [TC01] 查询所有记录
SELECT * FROM students;
-- 预期：返回 5 行数据，列包含 id, name, age, score

-- [TC02] 查询指定列
SELECT name, age FROM students;
-- 预期：返回 5 行，仅包含 name 和 age 两列

-- [TC03] SELECT 不存在的表
SELECT * FROM not_exist;
-- 预期：返回错误，提示表不存在

================================================================================
三、WHERE 条件查询
================================================================================

-- [TC04] 相等条件 =
SELECT * FROM students WHERE age = 22;
-- 预期：返回 1 行 (Bob)

-- [TC05] 大于/小于条件
SELECT * FROM students WHERE score > 85.0;
-- 预期：返回 Alice(1), Charlie, Diana, Alice(5) 共 4 行

-- [TC06] 逻辑与 AND
SELECT * FROM students WHERE age >= 20 AND score >= 90.0;
-- 预期：返回 Alice(1,20), Alice(5,23) 共 2 行

-- [TC07] 逻辑或 OR
SELECT * FROM students WHERE name = 'Bob' OR name = 'Charlie';
-- 预期：返回 2 行 (Bob, Charlie)

-- [TC08] 逻辑非 NOT
SELECT * FROM students WHERE NOT age = 22;
-- 预期：返回除了 Bob 外的 4 行

================================================================================
四、LIKE 模糊匹配
================================================================================

-- [TC09] LIKE 'A%' （以 A 开头）
SELECT * FROM students WHERE name LIKE 'A%';
-- 预期：返回两行 Alice (id=1) 和 Alice (id=5)

-- [TC10] LIKE '%e' （以 e 结尾）
SELECT * FROM students WHERE name LIKE '%e';
-- 预期：返回 Alice(id=1), Charlie, Alice(id=5) 三行

-- [TC11] LIKE 'A___e' （匹配 A 开头，e 结尾，中间 3 个字符）
SELECT * FROM students WHERE name LIKE 'A___e';
-- 预期：返回 Alice 的行

================================================================================
五、IN 与 BETWEEN
================================================================================

-- [TC12] IN 条件
SELECT * FROM students WHERE age IN (19, 21, 23);
-- 预期：返回 Charlie(19), Diana(21), Alice(23) 共 3 行

-- [TC13] BETWEEN 条件
SELECT * FROM students WHERE score BETWEEN 80.0 AND 90.0;
-- 预期：返回 Alice(88.5), Diana(85.5), Alice(90.0) 共 3 行 (90.0 包含)

================================================================================
六、ORDER BY 排序
================================================================================

-- [TC14] 升序排序
SELECT * FROM students ORDER BY age;
-- 预期：按照 age 从小到大排列 (Charlie, Alice(1), Diana, Bob, Alice(5))

-- [TC15] 降序排序
SELECT * FROM students ORDER BY score DESC;
-- 预期：按照 score 从大到小排列 (Charlie, Alice(5), Alice(1), Diana, Bob)

================================================================================
七、聚合函数
================================================================================

-- [TC16] COUNT(*)
SELECT COUNT(*) FROM students;
-- 预期：返回一行，值为 5

-- [TC17] AVG, MAX, MIN
SELECT AVG(score), MAX(age), MIN(age) FROM students;
-- 预期：AVG ≈ 86.4, MAX=23, MIN=19

-- [TC18] SUM
SELECT SUM(score) FROM students;
-- 预期：432.0 (88.5+76+92+85.5+90)

================================================================================
八、GROUP BY 与 HAVING
================================================================================

-- [TC19] 按姓名分组并统计人数
SELECT name, COUNT(*) AS cnt FROM students GROUP BY name HAVING cnt > 1;
-- 预期：返回 name='Alice', cnt=2

-- [TC20] 按年龄分组求平均分，筛选均分 > 85 的组
SELECT age, AVG(score) AS avg_score FROM students GROUP BY age HAVING avg_score > 85;
-- 预期：返回 age=19(92.0), age=20(88.5), age=23(90.0) 等（具体视数据）

================================================================================
九、DISTINCT 去重
================================================================================

-- [TC21] 查询不重复的姓名
SELECT DISTINCT name FROM students;
-- 预期：返回 Alice, Bob, Charlie, Diana

-- [TC22] 查询不重复的年龄
SELECT DISTINCT age FROM students ORDER BY age;
-- 预期：返回 19,20,21,22,23

================================================================================
十、LIMIT 与 OFFSET 分页
================================================================================

-- [TC23] 前 2 条记录
SELECT * FROM students ORDER BY id LIMIT 2;
-- 预期：返回 id=1,2 的记录

-- [TC24] 跳过前 3 条，取 2 条
SELECT * FROM students ORDER BY id LIMIT 2 OFFSET 3;
-- 预期：返回 id=4,5 的记录

-- [TC25] 超出范围的分页
SELECT * FROM students ORDER BY id LIMIT 10 OFFSET 100;
-- 预期：返回空结果，日志显示 "Selected 0 rows"

================================================================================
十一、INSERT 插入
================================================================================

-- [TC26] 插入一条指定列的数据
INSERT INTO students (id, name, age) VALUES (6, 'Eve', 25);
-- 预期：插入成功，查询 score 字段应为 NULL 或 0（取决于实现）

-- [TC27] 插入一行所有列的值（省略列名）
INSERT INTO students VALUES (7, 'Frank', 22, 78.5);
-- 预期：插入成功

-- [TC28] 插入多行
INSERT INTO students VALUES (8, 'Grace', 20, 88.0), (9, 'Hank', 21, 91.0);
-- 预期：插入两行

十二、UPDATE 更新
================================================================================

UPDATE students SET score = 95.0 WHERE name = 'Alice' AND age = 20;
-- 预期：预计更新 1 行 (id=1)

-- [TC30] 无条件更新（谨慎！）

-- [TC31] 更新不存在的记录
